# Pre-requisites (system-level):
#   - Python 3.9 – 3.11

#
# GPU support (optional but recommended):
#   Install PyTorch with CUDA first:
#     pip install torch torchaudio --index-url https://download.pytorch.org/whl/cu121
#   Then install the rest of the requirements.
# ============================================================

# --- Core ML / Audio ---
torch>=2.0.0
torchaudio>=2.0.0
# torchvision must match torch major version exactly:
#   torch 2.6.x -> torchvision 0.21.x
#   torch 2.7.x -> torchvision 0.22.x
#   torch 2.8.x -> torchvision 0.23.x
# Mismatches cause "operator torchvision::nms does not exist" at import.
# Install manually to match your torch build, e.g.:
#   pip install torchvision --index-url https://download.pytorch.org/whl/cpu
#   pip install torchvision --index-url https://download.pytorch.org/whl/cu121
torchvision
